129  {
130  this.obstacles[i] = new Rect(this.obstacles[i].X - this.speed, this.obstacles[i].Y, this.obstacles[i].Width, this.obstacles[i].Height);
131 
132  if (this.obstacles[i].X < 0 - (this.areaWidth / 8))
133  {
134  this.obstacles.RemoveAt(i);
135  }
136 
137  RectangleGeometry tempCarBody = new RectangleGeometry(this.carObj.CarBody)
138  {
139  Transform = new RotateTransform(this.carObj.Angle, this.carObj.CarBody.Left + (this.carObj.CarBody.Width / 2), this.carObj.CarBody.Top + (this.carObj.CarBody.Height / 2))
140  };
141 
142  Geometry collisionGeometry = Geometry.Combine(tempCarBody, new RectangleGeometry(this.obstacles[i]), GeometryCombineMode.Intersect, null);
143 
144  if (collisionGeometry.GetArea() != 0)
145  {
146  this.DoGameOver();
147  }
148  }
149 
150  if (this.obstacles.Last().X < this.areaWidth - (this.areaWidth / 4))
151  {
152  this.obstacles.Add(
153  new Rect(
154  this.r.Next(

167  public void StepLine(Direction direction)
168  {
169  this.stepCount++;
170 
171  if (this.previousDirection != direction || this.stepCount == this.speed * 7)
172  {
173  this.points.Add(new Point(this.points.Last().X, this.points.Last().Y));
174  this.stepCount = 0;
175  }
176 
177  for (int i = 0; i < this.points.Count - 1; i++)
178  {
179  if (this.points[i].X < 0 && this.points.Count(x => x.X < 0) > 2)
180  {
181  this.points.RemoveAt(i);
182  }
183  else
184  {
185  this.points[i] = new Point(this.points[i].X - this.speed, this.points[i].Y);
186  }
187  }
188 
189  switch (direction)
190  {
191  case Direction.Up:
192  if (this.points.Last().Y - this.carObj.CarBody.Height - this.carObj.WheelSize - (this.areaHeight / 20) > 0)
193  {
194  this.points[this.points.Count - 1] = new Point(this.points.Last().X, this.points.Last().Y - this.ySpeed);
195  }
196 
197  break;
198  case Direction.Down:
199  if (this.points.Last().Y + 5 < this.areaHeight - 45)
200  {
201  this.points[this.points.Count - 1] = new Point(this.points.Last().X, this.points.Last().Y + this.ySpeed);
202  }
203 
204  break;
205  default:
206  break;
207  }
208 
209  this.line = this.MakeCurve(this.points.ToArray(), 0.2);
210  this.previousDirection = direction;
211  }

223  private Point[] MakeCurvePoints(Point[] points, double tension)
224  {
225  if (points.Length < 2)
226  {
227  return null;
228  }
229 
230  double control_scale = tension / 0.5 * 0.175;
231 
232  // Make a list containing the points and
233  // appropriate control points.
234  List<Point> result_points = new List<Point>();
235  result_points.Add(points[0]);
236 
237  for (int i = 0; i < points.Length - 1; i++)
238  {
239  // Get the point and its neighbors.
240  Point pt_before = points[Math.Max(i - 1, 0)];
241  Point pt = points[i];
242  Point pt_after = points[i + 1];
243  Point pt_after2 = points[Math.Min(i + 2, points.Length - 1)];
244 
245  Point p4 = pt_after;
246 
247  double dx = pt_after.X - pt_before.X;
248  double dy = pt_after.Y - pt_before.Y;
249  Point p2 = new Point(
250  pt.X + (control_scale * dx),
251  pt.Y + (control_scale * dy));
252 
253  dx = pt_after2.X - pt.X;
254  dy = pt_after2.Y - pt.Y;
255  Point p3 = new Point(
256  pt_after.X - (control_scale * dx),
257  pt_after.Y - (control_scale * dy));
258 
259  // Save points p2, p3, and p4.
260  result_points.Add(p2);
261  result_points.Add(p3);
262  result_points.Add(p4);
263  }
264 
265  // Return the points.
266  return result_points.ToArray();
267  }

269  // Make a Path holding a series of Bezier curves.
270  // The points parameter includes the points to visit
271  // and the control points.
272  private PathGeometry MakeBezierPath(Point[] points)
273  {
274  // Add a PathGeometry.
275  PathGeometry path_geometry = new PathGeometry();
276 
277  // Create a PathFigure.
278  PathFigure path_figure = new PathFigure();
279  path_geometry.Figures.Add(path_figure);
280 
281  // Start at the first point.
282  path_figure.StartPoint = points[0];
283 
284  // Create a PathSegmentCollection.
285  PathSegmentCollection path_segment_collection =
286  new PathSegmentCollection();
287  path_figure.Segments = path_segment_collection;
288 
289  // Add the rest of the points to a PointCollection.
290  PointCollection point_collection =
291  new PointCollection(points.Length - 1);
292  for (int i = 1; i < points.Length; i++)
293  {
294  point_collection.Add(points[i]);
295  }
296 
297  // Make a PolyBezierSegment from the points.
298  PolyBezierSegment bezier_segment = new PolyBezierSegment();
299  bezier_segment.Points = point_collection;
300 
301  // Add the PolyBezierSegment to othe segment collection.
302  path_segment_collection.Add(bezier_segment);
303 
304  return path_geometry;
305  }
