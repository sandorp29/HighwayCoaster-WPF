104  private void GameArea_KeyDown(object sender, System.Windows.Input.KeyEventArgs e)
105  {
106  switch (e.Key)

108  case System.Windows.Input.Key.Up:
109  if (e.IsUp)
110  {
111  this.direction = Direction.None;
112  }
113  else
114  {
115  this.direction = Direction.Up;
116  }
117 
118  break;
119  case System.Windows.Input.Key.Down:
120  if (e.IsUp)
121  {
122  this.direction = Direction.None;
123  }
124  else
125  {
126  this.direction = Direction.Down;
127  }
128 
129  break;
130  case System.Windows.Input.Key.Escape:
131  this.Leave();
132  break;
133  default:
134  break;
135  }
136  }
